bubbleView = {
  if (!skillRows.length) return html`<p role="status">No skills match this search. Clear the search or try another term.</p>`;
  const control = viewof skillName;
  const peak = Math.max(1, ...scope.skills.map(d => skillMetric === "share" ? 100 * d.records / d.denominator : d[skillMetric]));
  const choose = name => {
    if (control.value === name) return;
    control.value = name;
    control.dispatchEvent(new Event("input", {bubbles: true}));
  };
  return html`<div class="skill-bubbles" role="group" aria-label="Skills by selected measure">${skillRows.map(d => {
    const diameter = 188 * Math.sqrt(d[skillMetric] / peak);
    const value = skillMetric === "share" ? d.share.toFixed(1) + "%" : d[skillMetric].toLocaleString("en-US");
    const button = html`<button type="button" class="ojs-bubble" data-skill=${d.name} style=${`--bubble-size:${diameter}px;--bubble-slot:${Math.max(100,diameter)}px`} aria-pressed="false" aria-controls="selected-skill-summary" aria-label=${`${d.name}: ${d.records} postings, ${d.companies} employers, ${d.share.toFixed(1)} percent. Show evidence.`}>
      <span class="bubble-disc" aria-hidden="true">${diameter >= 75 ? html`<strong>${value}</strong>` : ""}</span>
      <span class="bubble-caption">${d.name}${diameter < 75 ? html`<strong>${value}</strong>` : ""}</span>
    </button>`;
    button.addEventListener("click", () => choose(d.name));
    button.addEventListener("focus", () => choose(d.name));
    button.addEventListener("mouseenter", () => {
      if (!document.activeElement?.closest(".analytics-evidence, .skill-detail")) choose(d.name);
    });
    return button;
  })}</div>`;
}
